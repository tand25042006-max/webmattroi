// ============================================================ // HE THONG GIAM SAT SOLAR - TICH HOP TFT + WIFI + SENSOR // Phien bản: Full hỗ trợ đẩy trạng thái kết nối lên Web // ============================================================ #include #include #include #include #include #include #include #include #include #include #include #include #include #include // ─── CHÂN KẾT NỐI ────────────────────────────────────────── #define TFT_CS 5 #define TFT_DC 27 #define TFT_RST 26 #define ONE_WIRE_BUS 4 #define PIN_BUZZER 13 #define PIN_BTN_PAGE 32 #define PIN_BTN_MUTE 33 #define I2C_SDA 21 #define I2C_SCL 22 #define DEBOUNCE_MS 50 #define LONG_PRESS_MS 3000 // ─── WIFI & BACKEND ───────────────────────────────────────── const char* ssid = "HONOR 400 pro"; const char* password = " "; const char* serverTarget = "https://esp32-backend-vqfp.onrender.com/api/data"; const unsigned long WIFI_SEND_INTERVAL = 5000; unsigned long lastWifiSend = 0; // ─── KHỞI TẠO ĐỐI TƯỢNG ──────────────────────────────────── Adafruit_ILI9341 tft = Adafruit_ILI9341(TFT_CS, TFT_DC, TFT_RST); INA226_WE ina226 = INA226_WE(0x40); OneWire oneWire(ONE_WIRE_BUS); DallasTemperature ds18b20(&oneWire); Adafruit_MPU6050 mpu; Adafruit_VEML7700 veml; // ─── BIẾN INTERRUPT ──────────────────────────────────────── volatile bool btnPageEvent = false; volatile bool btnMuteEvent = false; volatile unsigned long btnPageDownTime = 0; // ─── ISR NÚT ─────────────────────────────────────────────── void IRAM_ATTR isr_btnPage() { if (digitalRead(PIN_BTN_PAGE) == LOW) { btnPageDownTime = millis(); } else { if ((millis() - btnPageDownTime) >= DEBOUNCE_MS) btnPageEvent = true; } } void IRAM_ATTR isr_btnMute() { static unsigned long lastMuteISR = 0; if (millis() - lastMuteISR >= DEBOUNCE_MS) { btnMuteEvent = true; lastMuteISR = millis(); } } // ─── BIẾN TOÀN CỤC ───────────────────────────────────────── int currentPage = 0; const int maxPages = 2; bool isCalibrated = false; bool hasError = false; bool errDS = false, errINA = false, errMPU = false, errVEML = false; bool muteBuzzer = false; String errorLog = ""; float g_tempDS, g_busVoltage, g_current_mA, g_lux, g_deltaPitch, g_deltaRoll, g_deltaYaw; unsigned long lastSensorRead = 0; // ─── HÀM WIFI & SERVER ────────────────────────────────────── void connectToWiFi() { WiFi.mode(WIFI_STA); WiFi.begin(ssid, password); int retry = 0; while (WiFi.status() != WL_CONNECTED && retry < 20) { delay(500); retry++; } } void sendDataToServer() { if (WiFi.status() != WL_CONNECTED) return; WiFiClientSecure client; client.setInsecure(); HTTPClient http; http.begin(client, serverTarget); http.addHeader("Content-Type", "application/json"); // JSON chứa cả giá trị đo và trạng thái kết nối StaticJsonDocument<512> doc; doc["dien_ap"] = errINA ? 0 : g_busVoltage; doc["dong_dien"] = errINA ? 0 : g_current_mA / 1000.0; doc["cong_suat"] = errINA ? 0 : (g_busVoltage * g_current_mA) / 1000.0; doc["nhiet_do"] = errDS ? 0 : g_tempDS; doc["anh_sang"] = errVEML ? 0 : g_lux; // Trạng thái (true = OK, false = Lỗi) doc["stt_ds18b20"] = !errDS; doc["stt_ina226"] = !errINA; doc["stt_mpu6050"] = !errMPU; doc["stt_veml7700"] = !errVEML; String jsonStr; serializeJson(doc, jsonStr); http.POST(jsonStr); http.end(); } // ─── SETUP ────────────────────────────────────────────────── void setup() { Serial.begin(115200); pinMode(PIN_BUZZER, OUTPUT); pinMode(PIN_BTN_PAGE, INPUT_PULLUP); pinMode(PIN_BTN_MUTE, INPUT_PULLUP); attachInterrupt(digitalPinToInterrupt(PIN_BTN_PAGE), isr_btnPage, CHANGE); attachInterrupt(digitalPinToInterrupt(PIN_BTN_MUTE), isr_btnMute, FALLING); SPI.begin(18, 19, 23, TFT_CS); tft.begin(); Wire.begin(I2C_SDA, I2C_SCL); ina226.init(); ds18b20.begin(); mpu.begin(0x68, &Wire); veml.begin(&Wire); connectToWiFi(); } // ─── LOOP ─────────────────────────────────────────────────── void loop() { unsigned long currentMillis = millis(); // Xử lý nút bấm... if (btnPageEvent) { btnPageEvent = false; currentPage = (currentPage + 1) % maxPages; tft.fillScreen(ILI9341_BLACK); } if (btnMuteEvent) { btnMuteEvent = false; muteBuzzer = true; } // Đọc cảm biến if (currentMillis - lastSensorRead >= 1000) { lastSensorRead = currentMillis; // Kiểm tra kết nối (Ping) errDS = (ds18b20.getTempCByIndex(0) == -127.0f); Wire.beginTransmission(0x40); errINA = (Wire.endTransmission() != 0); Wire.beginTransmission(0x68); errMPU = (Wire.endTransmission() != 0); Wire.beginTransmission(0x10); errVEML = (Wire.endTransmission() != 0); if (!errINA) { g_busVoltage = ina226.getBusVoltage_V(); g_current_mA = ina226.getCurrent_mA(); } if (!errVEML) { g_lux = veml.readLux(); } // ... (logic hiển thị màn hình TFT của bạn ở đây) } // Gửi dữ liệu if (currentMillis - lastWifiSend >= WIFI_SEND_INTERVAL) { lastWifiSend = currentMillis; sendDataToServer(); } }
